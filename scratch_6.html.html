async function sha256(str){
  const buf = new TextEncoder().encode(str);
  const hash = await crypto.subtle.digest('SHA-256', buf);
  return [...new Uint8Array(hash)].map(b => b.toString(16).padStart(2,'0')).join('');
}

function sanitizeHTML(html){
  const t = document.createElement('div');
  t.innerHTML = html;
  t.querySelectorAll('script,style,iframe,object,embed,link,meta').forEach(e => e.remove());
  t.querySelectorAll('*').forEach(el => {
    [...el.attributes].forEach(a => {
      if(/^on/i.test(a.name)) el.removeAttribute(a.name);
      if(a.name === 'href' && /^javascript:/i.test(a.value)) el.removeAttribute(a.name);
    });
  });
  return t.innerHTML;
}

async function doRegister(){
  const login = $('rLogin').value.trim().toLowerCase();
  const email = $('rEmail').value.trim().toLowerCase();
  const p1 = $('rPass').value, p2 = $('rPass2').value;
  const quest = $('rQuest').value;
  const ans = $('rAns').value.trim().toLowerCase();
  const err = $('rErr');
  err.textContent = '';

  if(login.length < 3)              return err.textContent = 'Логин минимум 3 символа';
  if(!/^[a-z0-9_]+$/.test(login))   return err.textContent = 'Логин: a-z, 0-9, _';
  if(!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) return err.textContent = 'Введите корректный email';
  if(p1.length < 4)                 return err.textContent = 'Пароль минимум 4 символа';
  if(p1 !== p2)                     return err.textContent = 'Пароли не совпадают';
  if(!ans)                          return err.textContent = 'Введите ответ на секретный вопрос';

  const users = loadUsers();
  if(users[login]) return err.textContent = 'Такой логин уже занят';
  if(Object.values(users).some(u => u.email === email))
    return err.textContent = 'Этот email уже используется';

  users[login] = {
    hash: await sha256(p1),
    email,
    quest,
    ansHash: await sha256(ans),
    created: Date.now()
  };
  saveUsers(users);
  localStorage.setItem(SESSION_KEY, login);
  startSession(login);
}

async function doLogin(){
  const raw = $('lLogin').value.trim().toLowerCase();
  const pass = $('lPass').value;
  const err = $('lErr');
  err.textContent = '';

  const users = loadUsers();
  // искать по логину или email
  let login = users[raw] ? raw : Object.keys(users).find(k => users[k].email === raw);
  if(!login) return err.textContent = 'Пользователь не найден';
  const hash = await sha256(pass);
  if(hash !== users[login].hash) return err.textContent = 'Неверный пароль';

  localStorage.setItem(SESSION_KEY, login);
  startSession(login);
}

function collectNotes(filter){
  const out = [];
  for(const cls of Object.keys(DB)){
    for(const subj of Object.keys(DB[cls])){
      for(const topic of Object.keys(DB[cls][subj])){
        DB[cls][subj][topic].forEach(n => {
          const item = {cls,subj,topic,...n};
          if(filter(item)) out.push(item);
        });
      }
    }
  }
  return out.sort((a,b) => new Date(b.date) - new Date(a.date));
}

function matchQuery(n, q){
  if(!q) return true;
  q = q.toLowerCase();
  const plain = (n.text||'').replace(/<[^>]+>/g,' ');
  return (n.title||'').toLowerCase().includes(q)
      || plain.toLowerCase().includes(q)
      || (n.tags ||[]).some(t => t.toLowerCase().includes(q))
      || n.topic.toLowerCase().includes(q)
      || n.subj.toLowerCase().includes(q);
}

let graphState = null;

function initGraph(){
  const canvas = $('graph');
  if(!canvas) return;
  const wrap = canvas.parentElement;
  const dpr = devicePixelRatio || 1;
  canvas.width  = wrap.clientWidth  * dpr;
  canvas.height = wrap.clientHeight * dpr;
  const ctx = canvas.getContext('2d');
  ctx.scale(dpr, dpr);

  const nodes = [], byId = {}, links = [];
  const add = (id,label,type,meta={}) => {
    if(byId[id]) return byId[id];
    const n = { id, label, type, ...meta,
      x: Math.random()*wrap.clientWidth, y: Math.random()*wrap.clientHeight,
      vx:0, vy:0, r: type==='note'?5 : type==='topic'?9 : type==='subject'?13 : 18 };
    nodes.push(n); byId[id] = n; return n;
  };
  const link = (a,b) => links.push({a: byId[a], b: byId[b]});

  for(const cls of Object.keys(DB)){
    add('c:'+cls, cls+' класс', 'class');
    for(const subj of Object.keys(DB[cls])){
      add('s:'+cls+':'+subj, subj, 'subject', {cls,subj});
      link('c:'+cls, 's:'+cls+':'+subj);
      for(const topic of Object.keys(DB[cls][subj])){
        add('t:'+cls+':'+subj+':'+topic, topic, 'topic', {cls,subj,topic});
        link('s:'+cls+':'+subj, 't:'+cls+':'+subj+':'+topic);
        for(const n of DB[cls][subj][topic]){
          add('n:'+n.id, n.title, 'note', {cls,subj,topic,id:n.id});
          link('t:'+cls+':'+subj+':'+topic, 'n:'+n.id);
          (n.tags||[]).forEach(tag => {
            add('tag:'+tag, '#'+tag, 'tag');
            link('n:'+n.id, 'tag:'+tag);
          });
        }
      }
    }
  }

  const W = wrap.clientWidth, H = wrap.clientHeight;
  let scale = 1, offsetX = 0, offsetY = 0;

  function tick(){
    for(let i=0;i<nodes.length;i++){
      for(let j=i+1;j<nodes.length;j++){
        const a=nodes[i], b=nodes[j];
        const dx=b.x-a.x, dy=b.y-a.y;
        const d2 = dx*dx+dy*dy || 1;
        if(d2 > 40000) continue;
        const d = Math.sqrt(d2), f = 400/d2;
        const fx = dx/d*f, fy = dy/d*f;
        a.vx -= fx; a.vy -= fy; b.vx += fx; b.vy += fy;
      }
    }
    for(const l of links){
      const dx = l.b.x-l.a.x, dy = l.b.y-l.a.y;
      const d = Math.hypot(dx,dy) || 1;
      const f = (d - 90) * 0.008;
      const fx = dx/d*f, fy = dy/d*f;
      l.a.vx += fx; l.a.vy += fy; l.b.vx -= fx; l.b.vy -= fy;
    }
    for(const n of nodes){
      n.vx += (W/2 - n.x) * 0.0008;
      n.vy += (H/2 - n.y) * 0.0008;
      n.vx *= 0.86; n.vy *= 0.86;
      n.x += n.vx; n.y += n.vy;
    }
  }

  function draw(){
    ctx.clearRect(0,0,W,H);
    ctx.save();
    ctx.translate(offsetX, offsetY);
    ctx.scale(scale, scale);
    ctx.strokeStyle = 'rgba(140,150,200,.22)';
    ctx.lineWidth = 1;
    for(const l of links){
      ctx.beginPath(); ctx.moveTo(l.a.x, l.a.y); ctx.lineTo(l.b.x, l.b.y); ctx.stroke();
    }
    const colors = { class:'#ffffff', subject:'#a8b4ff', topic:'#8ea2ff',
                     note:'#cfd5f5', tag:'#ffc48a' };
    for(const n of nodes){
      const isHover = graphState.hover === n;
      ctx.beginPath();
      ctx.arc(n.x, n.y, n.r + (isHover?3:0), 0, Math.PI*2);
      ctx.fillStyle = colors[n.type] || '#ccc';
      ctx.fill();
      if(isHover){ ctx.strokeStyle = '#4a6cf7'; ctx.lineWidth = 2; ctx.stroke(); }
    }
    ctx.font = '12px system-ui,sans-serif';
    ctx.fillStyle = 'rgba(220,225,255,.75)';
    for(const n of nodes){
      if(n.r < 9 && scale < 1.4 && graphState.hover !== n) continue;
      ctx.fillText(n.label, n.x + n.r + 5, n.y + 4);
    }
    ctx.restore();
  }

  function loop(){ tick(); draw(); requestAnimationFrame(loop); }
  let dragging=null, panning=false, lastX=0, lastY=0, downX=0, downY=0;
  const toWorld = e => {
    const r = canvas.getBoundingClientRect();
    return { x:(e.clientX - r.left - offsetX)/scale, y:(e.clientY - r.top - offsetY)/scale };
  };
  const pick = p => {
    for(let i=nodes.length-1;i>=0;i--){
      const n = nodes[i];
      if(Math.hypot(n.x-p.x, n.y-p.y) < n.r + 6) return n;
    }
    return null;
  };

  canvas.addEventListener('mousedown', e => {
    downX = e.clientX; downY = e.clientY;
    const p = toWorld(e), n = pick(p);
    if(n){ dragging = n; } else { panning = true; lastX = e.clientX; lastY = e.clientY; }
  });
  canvas.addEventListener('mousemove', e => {
    const p = toWorld(e), hit = pick(p);
    graphState.hover = hit;
    canvas.style.cursor = hit ? 'pointer' : (panning ? 'grabbing' : 'grab');
    if(dragging){ dragging.x = p.x; dragging.y = p.y; dragging.vx = dragging.vy = 0; }
    else if(panning){
      offsetX += e.clientX - lastX; offsetY += e.clientY - lastY;
      lastX = e.clientX; lastY = e.clientY;
    }
  });
  window.addEventListener('mouseup', e => {
    if(dragging && Math.hypot(e.clientX-downX, e.clientY-downY) < 4) openNode(dragging);
    dragging = null; panning = false;
  });
  canvas.addEventListener('wheel', e => {
    e.preventDefault();
    scale = Math.max(0.35, Math.min(3, scale * (e.deltaY < 0 ? 1.12 : 0.89)));
  }, {passive:false});

  let pinch = null;
  canvas.addEventListener('touchstart', e => {
    if(e.touches.length === 2){
      const [a,b] = e.touches;
      pinch = { d: Math.hypot(a.clientX-b.clientX, a.clientY-b.clientY), s: scale };
    } else if(e.touches.length === 1){
      const t = e.touches[0], p = toWorld(t), n = pick(p);
      if(n) dragging = n; else { panning = true; lastX = t.clientX; lastY = t.clientY; }
    }
  }, {passive:true});
  canvas.addEventListener('touchmove', e => {
    if(e.touches.length === 2 && pinch){
      const [a,b] = e.touches;
      const d = Math.hypot(a.clientX-b.clientX, a.clientY-b.clientY);
      scale = Math.max(0.35, Math.min(3, pinch.s * d / pinch.d));
    } else if(e.touches.length === 1){
      const t = e.touches[0];
      if(dragging){ const p = toWorld(t); dragging.x = p.x; dragging.y = p.y; }
      else if(panning){
        offsetX += t.clientX - lastX; offsetY += t.clientY - lastY;
        lastX = t.clientX; lastY = t.clientY;
      }
    }
  }, {passive:true});
  canvas.addEventListener('touchend', () => { dragging=null; panning=false; pinch=null; });

  graphState = { hover:null };
  loop();
}

function mergeDB(other){

  }
}

class Board {
  constructor(canvas){
    this.c = canvas; this.ctx = canvas.getContext('2d');
    this.color = '#1e2233'; this.size = 3; this.eraser = false;
    this.drawing = false; this.history = []; this.hasContent = false;
    this.reset(); this.bind();
  }
  reset(){
    const {ctx,c} = this;
    ctx.fillStyle = '#fff'; ctx.fillRect(0,0,c.width,c.height);
    ctx.lineCap = 'round'; ctx.lineJoin = 'round';
    this.history = [c.toDataURL()]; this.hasContent = false;
  }
  snapshot(){
    this.history.push(this.c.toDataURL());
    if(this.history.length > 30) this.history.shift();
  }
  undo(){
    if(this.history.length <= 1) return;
    this.history.pop();
    const img = new Image();
    img.onload = () => {
      const {ctx,c} = this;
      ctx.fillStyle = '#fff'; ctx.fillRect(0,0,c.width,c.height);
      ctx.drawImage(img,0,0);
    };
    img.src = this.history[this.history.length-1];
  }
  pos(e){
    const r = this.c.getBoundingClientRect();
    const p = e.touches ? e.touches[0] : e;
    return {
      x: (p.clientX - r.left) * (this.c.width  / r.width),
      y: (p.clientY - r.top ) * (this.c.height / r.height)
    };
  }
  start(e){ e.preventDefault(); this.drawing = true;
    const p = this.pos(e); this.ctx.beginPath(); this.ctx.moveTo(p.x, p.y); }
  move(e){
    if(!this.drawing) return; e.preventDefault();
    const p = this.pos(e);
    this.ctx.strokeStyle = this.eraser ? '#fff' : this.color;
    this.ctx.lineWidth = this.eraser ? this.size*4 : this.size;
    this.ctx.lineTo(p.x, p.y); this.ctx.stroke();
    this.hasContent = true;
  }
  end(){ if(!this.drawing) return; this.drawing = false; this.snapshot(); }
  bind(){
    const c = this.c;
    c.addEventListener('mousedown', e => this.start(e));
    c.addEventListener('mousemove', e => this.move(e));
    window.addEventListener('mouseup', () => this.end());
    c.addEventListener('mouseleave', () => this.end());
    c.addEventListener('touchstart', e => this.start(e), {passive:false});
    c.addEventListener('touchmove',  e => this.move(e),  {passive:false});
    c.addEventListener('touchend',   () => this.end());
  }
  toDataURL(){ return this.hasContent ? this.c.toDataURL('image/png') : null; }
}

  // если есть выделение — заменяем, иначе вставляем в каретку
  if(document.queryCommandSupported('insertHTML')){
    document.execCommand('insertHTML', false, html);
  } else {
    const sel = window.getSelection();
    if(!sel.rangeCount) return;
    const range = sel.getRangeAt(0);
    range.deleteContents();
    const frag = range.createContextualFragment(html);
    range.insertNode(frag);
  }
}

/* Предобработка: серый + контраст + ч/б + 2x масштаб */
async function preprocessImage(file, opts = {}){
  const scale = opts.scale || 2;
  const contrast = opts.contrast || 1.6;
  const doThreshold = opts.threshold !== false;

  return new Promise(resolve => {
    const img = new Image();
    img.onload = () => {
      const c = document.createElement('canvas');
      c.width  = img.width  * scale;
      c.height = img.height * scale;
      const ctx = c.getContext('2d');
      ctx.imageSmoothingEnabled = true;
      ctx.imageSmoothingQuality = 'high';
      ctx.drawImage(img, 0, 0, c.width, c.height);

      const imageData = ctx.getImageData(0, 0, c.width, c.height);
      const d = imageData.data;
      for(let i = 0; i < d.length; i += 4){
        let g = 0.299*d[i] + 0.587*d[i+1] + 0.114*d[i+2];
        g = Math.min(255, Math.max(0, (g - 128) * contrast + 128));
        if(doThreshold) g = g > 140 ? 255 : 0;
        d[i] = d[i+1] = d[i+2] = g;
      }
      ctx.putImageData(imageData, 0, 0);
      c.toBlob(b => resolve(b), 'image/png');
    };
    img.src = URL.createObjectURL(file);
  });
}

/* OCR через Tesseract — печатный */
async function ocrTesseract(blobOrFile, onProgress){
  const worker = await Tesseract.createWorker(['rus','eng'], 1, {
    logger: m => onProgress && onProgress(m)
  });
  // PSM 6 — единый блок текста; PSM 1 — авто; для рукописи хуже, но хоть что-то
  try {
    await worker.setParameters({
      tessedit_pageseg_mode: '1',
      preserve_interword_spaces: '1'
    });
  } catch(_){}
  const { data: { text } } = await worker.recognize(blobOrFile);
  await worker.terminate();
  return text;
}

/* OCR через TrOCR — рукописный (нейросеть) */
async function ocrTrOCR(file, onProgress){
  const pipeline = await window.__transformersReady;
  onProgress && onProgress('Загрузка модели TrOCR (может занять минуту)…');

  const ocr = await pipeline('image-to-text', 'Xenova/trocr-base-handwritten');

  // загрузить картинку как data URL и разбить на строки (упрощённо)
  const imgURL = await new Promise(res => {
    const r = new FileReader();
    r.onload = () => res(r.result);
    r.readAsDataURL(file);
  });

  onProgress && onProgress('Распознавание…');
  const result = await ocr(imgURL, { max_new_tokens: 64 });
  if(Array.isArray(result)) return result.map(r => r.generated_text).join('\n');
  return result[0]?.generated_text || result.generated_text || '';
}

  if(!SR) return alert('Браузер не поддерживает распознавание речи. Используйте Chrome/Edge.');
  if(recording){ stopRecognition(); return; }

  recognition = new SR();
  recognition.lang = 'ru-RU';
  recognition.continuous = true;
  recognition.interimResults = true;

  recognition.onresult = e => {
    let fin = '', int = '';
    for(let i=e.resultIndex; i<e.results.length; i++){
      const t = e.results[i][0].transcript;
      if(e.results[i].isFinal) fin += t + ' ';

  recognition.start();

  recBtn.classList.add('recording');
  recBtn.textContent = '⏹ Остановить запись';

/* Декодировать аудиофайл в Float32Array 16kHz mono */
async function decodeAudioToMono16k(file){
  const arrayBuffer = await file.arrayBuffer();
  const AC = window.AudioContext || window.webkitAudioContext;
  const ctx = new AC({ sampleRate: 16000 });
  const buf = await ctx.decodeAudioData(arrayBuffer.slice(0));
  // если stereo — усреднить
  if(buf.numberOfChannels > 1){
    const L = buf.getChannelData(0);
    const R = buf.getChannelData(1);
    const out = new Float32Array(L.length);
    for(let i=0;i<L.length;i++) out[i] = (L[i] + R[i]) / 2;
    return out;
  }
  return buf.getChannelData(0);
}
